ทำงานร่วมกับ ESP32 Input Pull-up (กดปุ่ม/วางมือถือ = LOW [0] | ปล่อยปุ่ม/ไม่วาง = HIGH [1])

โค้ดตัวอย่างสำหรับ ESP32 (Arduino IDE)

อัปโหลดโค้ดด้านล่างลงในบอร์ด ESP32 โดยต่อสวิตช์ทั้ง 10 ตัวเข้ากับขา GPIO ที่กำหนดและอีกขาต่อลง GND (ใช้ INPUT_PULLUP ภายในบอร์ด):

#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

WebServer server(80);

// กำหนดขา GPIO ทั้ง 10 ตัวสำหรับต่อ Switch (ต่อลง GND)
const int switchPins[10] = {13, 14, 27, 26, 25, 33, 32, 18, 19, 21};

void handleStatus() {
  // เปิด CORS เพื่อให้หน้าเว็บ HTML เรียกข้อมูลได้
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, OPTIONS");
  
  String json = "{\"switches\":[";
  for (int i = 0; i < 10; i++) {
    // INPUT_PULLUP: กดปุ่มค้าง (วางมือถือ) = LOW (0) -> ส่งค่า true
    bool isPressed = (digitalRead(switchPins[i]) == LOW);
    json += (isPressed ? "true" : "false");
    if (i < 9) json += ",";
  }
  json += "]}";
  
  server.send(200, "application/json", json);
}

void setup() {
  Serial.begin(115200);
  
  // ตั้งค่าขา Switch เป็น INPUT_PULLUP ทั้ง 10 ขา
  for (int i = 0; i < 10; i++) {
    pinMode(switchPins[i], INPUT_PULLUP);
  }

  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected! IP Address: ");
  Serial.println(WiFi.localIP());

  server.on("/status", HTTP_GET, handleStatus);
  server.begin();
}

void loop() {
  server.handleClient();
}
ข้อควรระวังเรื่อง Mixed Content (HTTPS / HTTP): หากเปิดไฟล์ HTML นี้ผ่านเว็บที่มี HTTPS เบราว์เซอร์อาจบล็อกการเชื่อมต่อไปยัง IP (HTTP) ของ ESP32 แนะนำให้เปิดไฟล์ index.html นี้จากในเครื่องคอมพิวเตอร์โดยตรง (Double-click เปิดไฟล์) หรืออนุญาต Insecure Content ในตั้งค่าเบราว์เซอร์
